        try {
          const txHash = await grantPermission();
          console.log("Grant Tx Hash:", txHash);

      callPermission();

  const handleSign = async (): Promise<void> => {
    if (!signingClient?.granteeAddress) {
      console.error("No grantee address to sign with.");
      return;

    try {
      const response = await signArb?.(

  const grantPermission = async () => {
    if (!signingClient || !bech32Address) throw new Error("Wallet not connected");

      // seconds: Math.floor(Date.now() / 1000 + 3600), // 1 hour from now

      nanos: 0,
    };
  
    const msgSendAuthorization = {
      typeUrl: "/cosmos.authz.v1beta1.MsgGrant",
      value: MsgGrant.encode({
        granter,
        grantee,
        grant: {
          authorization: {
            typeUrl: "/cosmos.bank.v1beta1.SendAuthorization",
            value: MsgSend.encode({
              amount:[{
                  denom: "uxion",
                  amount: "1000000" // Adjust as needed

              //   denom: "uxion",
              //   amount: "1000000" // Adjust as needed

      amount: [{ denom: "uxion", amount: "5000" }],
      gas: "200000",
    };
  
    const result = await signingClient.signAndBroadcast(granter, [msgSendAuthorization], fee);
    return result.transactionHash;
  };

  const transferTokens = async (
    recipientAddress: string,
    amount: Coin[],
    memo = ""
  ): Promise<string> => {
    if (!isConnected || !signingClient || !bech32Address) {
      throw new Error("Wallet not properly connected.");
    }

    try {
      return await attemptTransfer(recipientAddress, amount, memo);
    } catch (err: any) {
      const msg = err?.message || "";

      if (msg.includes("authorization not found")) {
        console.warn("No grant found. Attempting to grant permission...");

        await grantPermission();

        // Retry once after granting
        return await attemptTransfer(recipientAddress, amount, memo);
      }

      console.error("Transfer failed:", err);
      throw err;
    }
  };

  const attemptTransfer = async (
    to: string,
    amount: Coin[],
    memo = ""
  ): Promise<string> => {
    const result = await signingClient!.sendTokens(
      bech32Address!,
      to,
      amount,
      "auto",
      memo
    );
    if (result.code !== 0) {
      throw new Error(`Token transfer failed: ${result.rawLog}`);
    }
    return result.transactionHash;
  };

  const getMetaBalance = async (address: string, denom = "uxion") => {
    if (!queryClient) throw new Error("Query client is not initialized.");
    try {
      const balance = await queryClient.getBalance(address, denom);
      return balance?.amount ? uxionToXion(balance.amount) : "0";
    } catch (error) {
      console.error("Failed to fetch balance:", error);
      return "0";
    }
  };

  const getWalletQuery = async () => {
    if (!queryClient) throw new Error("Query client is not initialized.");
    try {
      const res = await queryClient.queryContractSmart("", { get_count: {} });
      console.log("Contract query result:", res);
      return res;
    } catch (err) {
      console.error("Smart contract query failed:", err);
      throw err;
    }
  };

  return {
    handleSign,
    grantPermission,
    transferTokens,
    getMetaBalance,
    getWalletQuery,
    signArbResponse,
  };
}
